Sales Header structure
CREATE TABLE SalesHeader (
    SalesOrder          VARCHAR(30)      PRIMARY KEY,
    SalesOrderDate      TIMESTAMP        NOT NULL,
    SalesOrderStatus    VARCHAR(20)      NOT NULL DEFAULT 'Open',
    SalesValue          DECIMAL(18,2)    NOT NULL,
    SalesLocalCurrency  VARCHAR(3)       NOT NULL,
    SalesTargetCurrency VARCHAR(3)       NOT NULL,
    ExchangeRate        DECIMAL(10,4),
    SalesRegion         VARCHAR(10)      NOT NULL,
    SalesOrganization   VARCHAR(10)      NOT NULL,
    SalesRep            VARCHAR(50),
    ProductID           VARCHAR(30)      NOT NULL,
    CustomerNumber      VARCHAR(30)      NOT NULL,
    CustomerName        VARCHAR(100)     NOT NULL,
    BillingAddress      VARCHAR(100),
    ShippingAddress     VARCHAR(100),
    City                VARCHAR(50),
    State               VARCHAR(50),
    Country             VARCHAR(50),
    PostalCode          VARCHAR(10),
    DeliveryMode        VARCHAR(20),
    DeliveryPartner     VARCHAR(50),
    Distributor         VARCHAR(50),
    Dealer              VARCHAR(50),
    Plant               VARCHAR(10),
    CompanyCode         VARCHAR(10),
    ExpectedDeliveryDate TIMESTAMP,
    ActualDeliveryDate   TIMESTAMP,
    CreatedBy           VARCHAR(50),
    LastModifiedAt      TIMESTAMP        DEFAULT CURRENT_TIMESTAMP
);
